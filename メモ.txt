■カラーアニメーション(フランス館)
●グラデーションアニメーション
int pts[] = primpoints(0, @primnum);

float values[];
foreach (int pt; pts)
{
    vector pos = point(0, "P", pt);
    append(values, pos.y);
}

// 高い順にソート
int order[] = argsort(values);

// id を設定
//sin(@Time * speed + offset - phase)
float time_cofficient = fit(sin(@Frame * 0.1), -1, 1, 0, 1);
float noise = fit(@noise, 0.4, 1, 0, 2);
float num = noise * 40;

for (int i = 0; i < num; i++)
{
    int pt = pts[order[i]];
    vector color = point(0, "Cd", pt);

    float gradation_cofficient = fit((num - i) / num, 0, time_cofficient, 0, 1);
    color.x += (1 - color.x) * gradation_cofficient;    
    color.y += (1 - color.y) * gradation_cofficient;
    color.z += (1 - color.z) * gradation_cofficient;
    setpointattrib(0, "Cd", pt, color, "set");
}
