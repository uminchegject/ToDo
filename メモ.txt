●グラデーションアニメーション
int pts[] = primpoints(0, @primnum);

float values[];
foreach (int pt; pts)
{
    vector pos = point(0, "P", pt);
    append(values, pos.y);
}

int order[] = argsort(values);
float pts_num = len(pts);

for (int i = 0; i < pts_num; i++)
{
    int pt = pts[order[i]];
    vector color = point(0, "Cd", pt);
    float gradation_cofficient = fit(sin(0.05 * i + @Frame * 0.1), -1, 1, 0, 1);
    color.x += (1 - color.x) * gradation_cofficient;    
    color.y += (1 - color.y) * gradation_cofficient;
    color.z += (1 - color.z) * gradation_cofficient;
    setpointattrib(0, "Cd", pt, color, "set");
}
