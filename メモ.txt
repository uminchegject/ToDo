float base_ratio = ch("base_ratio");
float unit_size = ch("unit_size");
float unit_ratio = ch("unit_ratio");
float offset_ratio = ch("offset_ratio");

vector base_scale = v@scale;
vector base_pos = v@P;

int unit_num = int(base_ratio * base_scale.y / unit_size);
for(int i = 0; i < unit_num; i++)
{
    vector pos = v@P;
    pos.y += (1 - base_ratio) * base_scale.y;
    pos.y += i  *  (base_ratio * base_scale.y / unit_num);
    
    vector scale = base_scale;
    scale.y = base_scale.y / unit_num;
    scale.y *= unit_ratio;
    scale.x *= offset_ratio;
    scale.z *= offset_ratio;
}

●set__height
float bounds[] = primintrinsic(0, "bounds", @primnum);
vector min_all = set(bounds[0], bounds[2], bounds[4]);
vector max_all = set(bounds[1], bounds[3], bounds[5]);
vector base_scale = max_all - min_all;
vector pos = v@P;

float unit_size = ch("unit_size");
float height = rand(v@P) * 150 + 30;

int unit_num_x = int(base_scale.x / unit_size);
int unit_num_z = int(base_scale.z / unit_size);
float unit_size_x = base_scale.x / unit_num_x;
float unit_size_z = base_scale.z / unit_num_z;

unit_size = unit_size_x;
if(unit_size_z < unit_size_x) unit_size = unit_size_z;
vector scale = set(unit_size, height, unit_size);

for(int x_idx = 0; x_idx < unit_num_x; x_idx++)
{
    for(int z_idx = 0; z_idx < unit_num_z; z_idx++)
    {
        vector spawn_pos = pos;
        spawn_pos.x += base_scale.x / 2 - (x_idx + 0.5) * unit_size_x;
        spawn_pos.z += base_scale.z / 2 - (z_idx + 0.5) * unit_size_z;

        int pt = addpoint(0, spawn_pos);
        setpointattrib(0, "scale", pt, scale);

        if ((x_idx + z_idx) % 2 == 0){
            setpointgroup(0, "PipeBlock", pt, 1);
        }

    }
}
removeprim(0, i@primnum, 1);


●make__Base_Basepoint
vector base_scale = v@scale;

float unit_size = base_scale.x / 6;

float height_unit_size = ch("height_unit_size");
int height_unit_num = int(base_scale.y / height_unit_size);
height_unit_size = base_scale.y / height_unit_num;

for(int idx = 0; idx < height_unit_num; idx++)
{
    vector pos = v@P;
    pos.y += height_unit_size * idx;
    
    //Frame
    vector frame_scale = set(unit_size * 2, height_unit_size, unit_size * 2);
    int frame_pt = addpoint(0, pos);
    setpointattrib(0, "scale", frame_pt, frame_scale);
    
    //Panel
    if((height_unit_num - idx) < 5 + rand(v@P) * 10){
        vector panel_scale = set(unit_size * 2, height_unit_size, unit_size * 2);
        int panel_pt = addpoint(0, pos);
        setpointattrib(0, "scale", panel_pt, panel_scale);
        setpointgroup(0, "panel", panel_pt, 1);
    }
}
removepoint(0, @ptnum);
