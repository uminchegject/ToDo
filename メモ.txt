○グラデーションアニメーション
float time_cofficient = sin(@Time);
int num = noise * 40;


for (int i = 0; i < num; i++)
{
    int pt = pts[order[i]];
    vector color = point(0, "Cd", pt);

    float gradation_cofficient = fit(float(num - i) / num, 0, noise_cofficient, 0, 1);
    color.x += (1 - color.x) * gradation_cofficient * time_cofficient;    
    color.y += (1 - color.y) * gradation_cofficient * time_cofficient;
    color.z += (1 - color.z) * gradation_cofficient * time_cofficient;
    setpointattrib(0, "Cd", pt, color, "set");
}
