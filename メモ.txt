block_scale.x *= (1.0 - center_ratio) * ratio + center_ratio;
block_scale.z *= (1.0 - center_ratio) * ratio + center_ratio;

block_scale.y *= 1.0 -
    ratio * (rand(idx + 10000) * 0.6 + 0.7);

int volume_class = int(random(idx) * 2);

// --------------------------------------------------

import Rhino.Geometry as rg


def panelize_surface(S, nu, nv):

    # U/V Domain
    u_domain = S.Domain(0)
    v_domain = S.Domain(1)

    u0 = u_domain.T0
    u1 = u_domain.T1

    v0 = v_domain.T0
    v1 = v_domain.T1

    # U/V方向の分割位置
    u_values = [
        u0 + (u1 - u0) * i / float(nu)
        for i in range(nu + 1)
    ]

    v_values = [
        v0 + (v1 - v0) * j / float(nv)
        for j in range(nv + 1)
    ]

    panels = []

    for i in range(nu):
        for j in range(nv):

            u_interval = rg.Interval(
                u_values[i],
                u_values[i + 1]
            )

            v_interval = rg.Interval(
                v_values[j],
                v_values[j + 1]
            )

            # 元Surfaceを複製
            #panel = S.DuplicateSurface()

            # UV範囲でトリム
            panel = S.Trim(
                u_interval,
                v_interval
            )

            if panel is not None:
                panels.append(panel)

    return panels


# ============================================================
# 2. Panel中心のUV座標を取得
# ============================================================

def get_panel_center_uv(panel):

    u_domain = panel.Domain(0)
    v_domain = panel.Domain(1)

    # UV Domainの中心
    u = (u_domain.T0 + u_domain.T1) / 2.0
    v = (v_domain.T0 + v_domain.T1) / 2.0

    return u, v


# ============================================================
# 3. Panel中心でPrincipal Curvatureを取得
# ============================================================

def get_principal_curvature(panel):

    # Panel中心のUV座標
    u, v = get_panel_center_uv(panel)

    # Curvature情報を取得
    curvature = panel.CurvatureAt(u, v)

    # 第1主曲率
    k1 = curvature.Kappa(0)

    # 第2主曲率
    k2 = curvature.Kappa(1)

    return k1, k2


# ============================================================
# 4. 全PanelのPrincipal Curvatureを計算
# ============================================================

def calculate_principal_curvatures(panels):

    k1_list = []
    k2_list = []

    for panel in panels:

        k1, k2 = get_principal_curvature(panel)

        k1_list.append(k1)
        k2_list.append(k2)

    return k1_list, k2_list


# ============================================================
# 5. Gaussian Curvatureを計算
#
# K = k1 * k2
# ============================================================

def calculate_gaussian_curvature(k1_list, k2_list):

    gaussian_curvature = []

    for k1, k2 in zip(k1_list, k2_list):

        K = k1 * k2

        gaussian_curvature.append(K)

    return gaussian_curvature


# ============================================================
# 6. Mean Curvatureを計算
#
# H = (k1 + k2) / 2
# ============================================================

def calculate_mean_curvature(k1_list, k2_list):

    mean_curvature = []

    for k1, k2 in zip(k1_list, k2_list):

        H = (k1 + k2) / 2.0

        mean_curvature.append(H)

    return mean_curvature


# ============================================================
# 7. Main処理
# ============================================================

def main(surface, nu, nv):

    # --------------------------------------------------------
    # STEP 1
    # SurfaceをPanel化
    # --------------------------------------------------------

    panels = panelize_surface(
        surface,
        nu,
        nv
    )

    # --------------------------------------------------------
    # STEP 2
    # Principal Curvature
    # --------------------------------------------------------

    k1_list, k2_list = calculate_principal_curvatures(
        panels
    )

    # --------------------------------------------------------
    # STEP 3
    # Gaussian Curvature
    # --------------------------------------------------------

    gaussian_curvature = calculate_gaussian_curvature(
        k1_list,
        k2_list
    )

    # --------------------------------------------------------
    # STEP 4
    # Mean Curvature
    # --------------------------------------------------------

    mean_curvature = calculate_mean_curvature(
        k1_list,
        k2_list
    )

    # --------------------------------------------------------
    # 結果をまとめて返す
    # --------------------------------------------------------

    return (
        panels,
        k1_list,
        k2_list,
        gaussian_curvature,
        mean_curvature
    )


# ============================================================
# Main
# ============================================================

(

    MeanCurvature
) = main(
    S,
    nu,
