HAPI_Result result;

HAPI_Session Session;
HAPI_CookOptions CookOptions;
HAPI_SessionInfo sessionInfo;
CookOptions = HAPI_CookOptions_Create();
HAPI_CreateInProcessSession(&Session, &sessionInfo);
HAPI_Initialize(&Session,&CookOptions,true,-1,nullptr,nullptr,nullptr,nullptr,nullptr);

HAPI_NodeId nodeId;
result = HAPI_CreateNode(&Session, -1, "Sop/box", "Box_Node_Name", true, &nodeId);

std::string geo_path = "C:/Users/yusuke.nakamura/Desktop/box_geo.geo";
result = HAPI_LoadGeoFromFile(&Session, nodeId, geo_path.c_str());

HAPI_GeoInfo geoInfo;
result = HAPI_GetGeoInfo(&Session, nodeId, &geoInfo);
if (result)
{
	HAPI_Cleanup(&Session);
}

// 3. パーツ情報取得
HAPI_PartInfo part_info;
checkHapiResult(HAPI_GetPartInfo(&session, geo_node, 0, &part_info));
std::cout << "Point Count: " << part_info.pointCount << std::endl;

// 4. Point座標の取得
HAPI_AttributeInfo attrib_info;
checkHapiResult(HAPI_GetAttributeInfo(&session, geo_node, 0, "P", HAPI_AttributeOwner::HAPI_ATTROWNER_POINT, &attrib_info));

// 5. メッシュデータ取得
std::vector<int> vertex_list(part_info.vertexCount);
checkHapiResult(HAPI_GetVertexList(&session, geo_node, 0, vertex_list.data(), 0, part_info.vertexCount));

std::vector<int> face_counts(part_info.faceCount);
checkHapiResult(HAPI_GetFaceCounts(&session, geo_node, 0, face_counts.data(), 0, part_info.faceCount));

// 6. 後処理
checkHapiResult(HAPI_Cleanup(&session));
checkHapiResult(HAPI_CloseSession(&session));
