○LineToBlock
vector blockSize = (1.0, 1.0, 1.0);

int primPtsList = primpoints(i@primnum);
for(int i = 0; i < len(primPtsList) - 1; i++)
{
	int pt1 = primPtsList[i];
	int pt2 = primPtsList[i + 1];
	vector pt1Pos = point(0, "P", pt1);
	vector pt2Pos = point(0, "P", pt2);
	
	int widthBlockNum = int(length(pt2Pos - pt1Pos) / blockSize.x);
	int heightBlockNum = int(@Height / blockSize.y);
	
	vector widthSlider = (pt2Pos - pt1Pos) / widthBlockNum;
	vector heightSlider;
	heightSlider.y = heightBlockNum;
	
	for(int width = 0; width < widthBlockNum; width++)
	{	
		for(int height = 0; height < heightBlockNum; height++)
		{
			vector spawnPos = pt1Pos;
			spawnPos += width * widthSlider;
			spawnPos += height * heightSlider;
			
			int pt = addpoint(0, spawnPos);
		}
	}
}
