def SetTPose():
    rootList = cmds.ls(assemblies = True)
    
    jointList = set()
    for root in rootList:
    	children = set(cmds.listRelatives(rootList, fullPath=True, type = "joint") or [])
    	while children:
    		jointList.update(children)
    		children = set(cmds.listRelatives(children, fullPath=True, type = "joint") or []) - jointList
    
    for joint in jointList:
        print(joint)
        cmds.setAttr( joint + '.rotate', 0, 0, 0, type = 'double3' )


def SetBindPose():
    rootList = cmds.ls(assemblies = True)
    for root in rootList:
        pose = cmds.dagPose(root, q = 1, bp = 1)			
        if pose:
            cmds.dagPose(root, r = 1, g = 1, bp = 1)
