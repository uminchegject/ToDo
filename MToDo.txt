○BaseSideSurfaceからBasePointの生成
vector blockSize = chv("BlockSize");

int primPts[] = primpoints(0, i@primnum);
int basePt = primPts[0];
int heightPts[];
int widthPts[];


vector heightPt1 = point(0, "P", heightPts[0]);
vector heightPt2 = point(0, "P", heightPts[1]);
vector widthPt1 = point(0, "P", widthPts[0]);
vector widthPt2 = point(0, "P", widthPts[1]);

vector blockSurfaceSize;
blockSurfaceSize.x = length(widthPt2 - widthPt1);
blockSurfaceSize.y = length(heightPt2 - heightPt1);

widthPtNum = int(blockSurfaceSize.x / blockSize.x);
heightPtNum = int(blockSurfaceSize.y / blockSize.y);

vector slider;
slider_x = (blockSurfaceSize.x / widthPtNum) * normal(widthPt2 - widthPt1);
slider_y = (blockSurfaceSize.y / heightPtNum) * normal(heightPt2 - heightPt1);

for(int width = 0; width < widthPtNum; ++width)
{
	for(int height = 0; height < heightPtNum; ++height)
	{
		vector spawnPos = v@P;
		spawnPos += width * slider_x - (slider_x * width) / 2;
		spawnPos -= height * slider_y - (slider_y * height) / 2;
		
		addpoint(0, spawnPos);
	}
}

○Normal
// 法線 (N) を正規化
vector N = normalize(@N);

// アップベクトルを定義（任意の固定方向）
vector up = {0, 1, 0};

// N と平行な場合を避けるため、適宜調整
if (abs(dot(N, up)) > 0.99) {
    up = {0, 0, 1};
}

// タンジェントベクトルを計算
vector tangent = normalize(cross(up, N));

// バイノーマルを計算
vector binormal = normalize(cross(N, tangent));

// 3x3の回転行列を作成
matrix3 rotationMatrix = set(tangent, binormal, N);

// 回転行列から四元数を生成
@orient = quaternion(rotationMatrix);
