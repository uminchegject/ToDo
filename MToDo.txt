○GreebleにTypeを仕込む
・TypeとサイズのMapを入れる
 -Folderパラメーターで対応
・サイズに適応したTypeがあった場合にTypeを設定し、Greebleを中断する
int vivitAssetToggle = 0;
for(int paramIndex = 0; paramIndex < ch("../Greeble/TypeList"); paramIndex++)
{
	vector scaleRange = chv("../Greeble/greeble_AddType" + itoa(paramIndex));
	vector type = chi("../Greeble/greeble_AddTypeScaleRange" + itoa(paramIndex));
	vector ratio = ch("../ParamInput/ratio" + itoa(paramIndex));

	float minLength = dividingBlockScaleOrigin.x;
	if(dividingBlockScaleOrigin.y < minLength) minLength = dividingBlockScaleOrigin.y;
	if(dividingBlockScaleOrigin.z < minLength) minLength = dividingBlockScaleOrigin.z;
	
	if(random(dividingBlockPosOrigin) < ratio) continue;	
	if(minLength < scaleRange.x) continue;
	if(scaleRange.y < minLength) continue;
	
	vivitAssetToggle = 1;
	break;
}

if(vivitAssetToggle == 1)
{
    int pt = addpoint(0, dividingBlockPosOrigin);
    setpointattrib(0, "scale", pt, dividingBlockScaleOrigin);
    setpointattrib(0, "orient", pt, p@orient);
	continue;
}
