○geoからの情報の取得方法を調査
void parseGeoFile(const std::string& filePath) {
    std::ifstream file(filePath);
    if (!file.is_open()) {
        std::cerr << "Failed to open file: " << filePath << std::endl;
        return;
    }

    std::string line;
    while (std::getline(file, line)) {
        // ファイルの行を解析する
        if (line.find("Point") != std::string::npos) {
            std::cout << "Point data found: " << line << std::endl;
        } else if (line.find("Primitive") != std::string::npos) {
            std::cout << "Primitive data found: " << line << std::endl;
        }
        // 他の構造も必要に応じて処理
    }

    file.close();
}
