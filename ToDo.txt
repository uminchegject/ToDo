import Rhino.Geometry as rg
# =========================================================
# 入力
# Srf       : 自由曲面
# UCount    : U方向の分割数
# VCount    : V方向の分割数
# Tolerance : 平面とみなす許容誤差
# =========================================================


# ---------------------------------------------------------
# 曲面をパネルに分割
# ---------------------------------------------------------

def divide_surface(srf, u_count, v_count):

    panels = []

    u_domain = srf.Domain(0)
    v_domain = srf.Domain(1)

    for i in range(u_count):

        for j in range(v_count):

            u0 = u_domain.ParameterAt(float(i) / u_count)
            u1 = u_domain.ParameterAt(float(i + 1) / u_count)

            v0 = v_domain.ParameterAt(float(j) / v_count)
            v1 = v_domain.ParameterAt(float(j + 1) / v_count)

            panel = srf.Trim(
                rg.Interval(u0, u1),
                rg.Interval(v0, v1)
            )

            panels.append(panel)

    return panels


# ---------------------------------------------------------
# パネルが平面に近いか判定
# ---------------------------------------------------------

def check_planarity(panel, tolerance):

    corners = [
        panel.PointAt(panel.Domain(0).T0, panel.Domain(1).T0),
        panel.PointAt(panel.Domain(0).T1, panel.Domain(1).T0),
        panel.PointAt(panel.Domain(0).T1, panel.Domain(1).T1),
        panel.PointAt(panel.Domain(0).T0, panel.Domain(1).T1)
    ]

    plane = rg.Plane.FitPlaneToPoints(corners)[1]

    max_error = 0

    for point in corners:

        error = abs(plane.DistanceTo(point))

        max_error = max(max_error, error)

    return max_error <= tolerance, max_error


# ---------------------------------------------------------
# パネルを評価
# ---------------------------------------------------------

def evaluate_panels(panels, tolerance):

    results = []

    for panel in panels:

        is_planar, error = check_planarity(
            panel,
            tolerance
        )

        if is_planar:
            panel_type = "Planar"
            cost = 100

        else:
            panel_type = "Curved"
            cost = 200

        results.append({
            "panel": panel,
            "type": panel_type,
            "error": error,
            "cost": cost
        })

    return results


# ---------------------------------------------------------
# 実行
# ---------------------------------------------------------

panels = divide_surface(
    Srf,
    UCount,
    VCount
)

results = evaluate_panels(
    panels,
    Tolerance
)


# ---------------------------------------------------------

PanelGeometry = [
