float min_length = ch("MinLength");
int prim_points[] = primpoints(0, @primnum);
for(int i = 1; i < len(prim_points); i++)
{
    vector pt_1_pos = point(0, "P", prim_points[i - 1]);
    vector pt_2_pos = point(0, "P", prim_points[i]);
    
    if(length(pt_2_pos - pt_1_pos) < min_length) 
    {
        removeprim(0, @primnum, 1);
        break;
    }
}
