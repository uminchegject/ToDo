■ソースfloat block_length_range = ch("block_length_range");
vector base_scale = chv("base_scale");
vector adjust_scale = chv("adjust_scale");
vector base_rotate = chv("base_rotate");
vector adjust_rotate = chv("adjust_rotate");
vector base_transform = chv("base_transform");
vector adjust_transform = chv("adjust_transform");

int prim_points[] = primpoints(0, @primnum);
int start_pt = prim_points[0];
int end_pt = prim_points[1];
vector start_pt_pos = point(0, "P", start_pt);
vector end_pt_pos = point(0, "P", end_pt);

float base_line_length = length(start_pt_pos - end_pt_pos);
vector line_dir = normalize(end_pt_pos - start_pt_pos);
vector normal = normalize(v@N);
vector tangent = normalize(line_dir - dot(line_dir, normal) * normal);
vector binormal = cross(normal, tangent);
vector4 q_line_dir = quaternion(set(binormal, tangent, normal));
matrix3 m_transform = set(tangent, normal, binormal);

float block_length = block_length_range.x + (block_length_range.y - block_length_range.x) / 2;
int base_point_num = int(base_line_length / block_length);
if(base_point_num == 0) base_point_num = 1;
float fixed_block_length = base_line_length / base_point_num;

for(int i = 0; i < base_point_num; ++i)
{
    vector spawn_pos = start_pt_pos;
    spawn_pos +=  line_dir * fixed_block_length * (i + 0.5);
    vector transform = base_transform + random(spawn_pos) * adjust_transform;
    spawn_pos += transform * m_transform;
    int pt = addpoint(0, spawn_pos);
    
    vector scale = base_scale + random(spawn_pos) * adjust_scale;
    scale.y *= fixed_block_length;
    setpointattrib(0, "scale", pt, scale);
    
    vector rotate = base_rotate + random(spawn_pos * 10) * adjust_rotate;
    rotate = radians(rotate);
    vector4 q_rotate = eulertoquaternion(rotate, 0);
    q_rotate = qmultiply(q_line_dir, q_rotate);
    setpointattrib(0, "orient", pt, q_rotate);
}

removeprim(0, @primnum, 1);
