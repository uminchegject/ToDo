{       
    float ratio = pt_idx / pt_num;

    scale = fit(scale, 0, gradation_ratio, 0, 1);

}

●揺れ
float amp = chf("amp");
float speed = chf("speed");
float delay = chf("delay");

int prim_pts[] = primpoints(0, @primnum);

for(int pt_idx = 0; pt_idx < len(prim_pts); ++pt_idx)
{

    float u = float(pt_idx) / float(len(prim_pts) - 1);
    float weight = pow(u, 2.0);
    float offset = rand(@primnum) * M_PI * 2.0;
    float phase = u * delay;

    float angle =
            sin(@Time * speed + offset - phase)
            * amp
            * weight;

    vector pivot = point(0, "P", prim_pts[0]);
    vector rel = point(0, "P", prim_pts[pt_idx]);
    rel -= pivot;

    matrix3 rot = ident();
    rotate(rot, angle, {0,0,1});
    rel *= rot;
    vector pos = pivot + rel;
    setpointattrib(0, "P", prim_pts[pt_idx], pos);
}
