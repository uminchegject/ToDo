●generate_simple_base_point
void generate_edge_base_point(vector pos; vector dir_1; vector dir_2;  float unit_size; vector color)
{
	vector normal = dir_2;
	
	vector pt1_pos = pos;
	pt1_pos += dir_1 * unit_size;
	pt1 = addpoint(0, pt1_pos);
	setpointattrib(0, "scale", pt1, set(unit_size, unit_size, unit_size * 2), "set");
	setpointattrib(0, "N", pt1, normal, "set");
	setpointattrib(0, "color", pt1, color, "set");
	
	vector pt2_pos = pos;
	pt2_pos -= dir_1 * unit_size;
	pt2_pos += dir_2 * unit_size;
	pt2 = addpoint(0, pt2_pos);
	setpointattrib(0, "scale", pt2, set(unit_size, unit_size, unit_size * 2), "set");
	setpointattrib(0, "N", pt2, normal, "set");
	setpointattrib(0, "color", pt2, color, "set");

}


●揺れ
float amp = chf("amp");
float speed = chf("speed");
float delay = chf("delay");

float u = float(@ptnum) / float(@numpt - 1);
float weight = pow(u, 2.0);
float offset = rand(@primnum) * M_PI * 2.0;
float phase = u * delay;

float angle =
    sin(@Time * speed + offset - phase)
    * amp
    * weight;

vector pivot = point(0, "P", 0);
vector rel = @P - pivot;
matrix3 rot = ident();
rotate(rot, angle, {0,0,1});
rel *= rot;
@P = pivot + rel;
