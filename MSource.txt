●ミラノ万博中国館の形態合理化
import Rhino.Geometry as rg


# --------------------------------------------------
# 1. 稜線を取得
# --------------------------------------------------

def get_ridges(surface):

    brep = surface.ToBrep()

    return [
        edge.DuplicateCurve()
        for edge in brep.Edges
    ]


# --------------------------------------------------
# 2. 稜線を平行な直線にする
# --------------------------------------------------

def make_parallel_lines(curves):

    if not curves:
        return []

    # 全体の基準方向
    direction = (
        curves[0].PointAtEnd
        - curves[0].PointAtStart
    )

    direction.Unitize()

    result = []

    for curve in curves:

        center = curve.PointAt(
            curve.Domain.Mid
        )

        length = curve.GetLength()

        p0 = center - direction * length * 0.5
        p1 = center + direction * length * 0.5

        result.append(
            rg.LineCurve(p0, p1)
        )

    return result


# --------------------------------------------------
# 3. 中央の2本を基準線にする
# --------------------------------------------------

def get_reference_lines(lines, surface):

    center = surface.GetBoundingBox(
        True
    ).Center

    lines = sorted(
        lines,
        key=lambda x:
        x.PointAt(x.Domain.Mid).DistanceTo(center)
    )

    return lines[:2]


# --------------------------------------------------
# 4. 2m間隔のグリッドを作る
# --------------------------------------------------

def make_grid(surface, spacing):

    bbox = surface.GetBoundingBox(True)

    min_x = bbox.Min.X
    max_x = bbox.Max.X
    min_y = bbox.Min.Y
    max_y = bbox.Max.Y

    lines = []

    x = min_x

    while x <= max_x:

        p0 = rg.Point3d(
            x,
            min_y,
            bbox.Min.Z
        )

        p1 = rg.Point3d(
            x,
            max_y,
            bbox.Max.Z
        )

        lines.append(
            rg.LineCurve(p0, p1)
        )

        x += spacing

    return lines


# --------------------------------------------------
# 5. 境界をグリッドに合わせる
# --------------------------------------------------

def snap_boundary(surface, spacing):

    bbox = surface.GetBoundingBox(True)

    curves = []

    for edge in surface.ToBrep().Edges:

        p0 = edge.PointAtStart
        p1 = edge.PointAtEnd

        # Y方向の位置を2m単位にする
        y0 = round(p0.Y / spacing) * spacing
        y1 = round(p1.Y / spacing) * spacing

        p0 = rg.Point3d(
            p0.X,
            y0,
            p0.Z
        )

        p1 = rg.Point3d(
            p1.X,
            y1,
            p1.Z
        )

        curves.append(
            rg.LineCurve(p0, p1)
        )

    return curves


# --------------------------------------------------
# 6. 境界を直線・円弧にする
# --------------------------------------------------

def rationalize_boundary(curves):

    result = []

    for curve in curves:

        points = [
            curve.PointAtStart,
            curve.PointAt(
                curve.Domain.Mid
            ),
            curve.PointAtEnd
        ]

        # 3点から円弧を作成
        arc = rg.Arc(
            points[0],
            points[1],
            points[2]
        )

        if arc.IsValid:
            result.append(
                rg.ArcCurve(arc)
            )
        else:
            result.append(
                rg.LineCurve(
                    points[0],
                    points[2]
                )
            )

    return result

# ==================================================
