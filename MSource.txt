●generate__obj_base_point
float pattern_height = detail(0, "PatternInterval");
vector axis_vec = set(0, 1, 0);

int prim_points[] = primpoints(0, i@primnum);

for(int i = 1; i < len(prim_points); ++i)
{
    for(int param_index = 1; param_index < ch("../ObjList") + 1; param_index++)
    {
        vector obj_scale = chv("../ObjScale" + itoa(param_index));
        vector obj_transform = chv("../ObjTransform" + itoa(param_index));
        vector obj_color = chv("../ObjColor" + itoa(param_index));
        float block_width = ch("../ObjBlockWidth" + itoa(param_index));

        vector last_pt_pos = point(0, "P", prim_points[i - 1]);
        vector next_pt_pos = point(0, "P", prim_points[i]);             
        
        int block_num = int(length(next_pt_pos - last_pt_pos) / block_width);
        if(block_num == 0) block_num = 1;
        
        vector slide_vector = (next_pt_pos - last_pt_pos) / block_num;  
        block_width = length(slide_vector);
                
        vector scale = set(obj_scale.x * length(slide_vector), obj_scale.y * pattern_height, obj_scale.z);
                
        int prim;
        vector uv;
        xyzdist(1, (next_pt_pos - last_pt_pos) / 2 + last_pt_pos, prim, uv);
        vector normal = prim(1, "N", prim);
    
        vector base_pos = last_pt_pos + slide_vector / 2;
        base_pos += normal * obj_transform.z;
        base_pos += slide_vector * obj_transform.x * block_width;
        base_pos += axis_vec * obj_transform.y * pattern_height;
        
        for(int j = 0; j < block_num; ++j)
        {
            vector pos = base_pos + slide_vector * j;
            int pt = addpoint(0, pos);
            setpointattrib(0, "scale", pt, scale);
            setpointattrib(0, "N", pt, normal);
            setpointattrib(0, "Cd", pt, obj_color);
        }
    }       
}

removeprim(0, i@primnum, 1);
