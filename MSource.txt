●ソース
float rotate_angle = detail(-1, "RotateAngleX");
float place_angle = radians(rotate_angle);
float line_angle  = radians(rotate_angle + 90);
int index = detail(4, "iteration");
float min_interval = detail(-1, "MinInterval", 0);
float max_interval = detail(-1, "MaxInterval", 0);
float interval_range = max_interval - min_interval;
float base_interval = min_interval + interval_range / 2;

function vector rotate2D(vector v; float ang)
{
    return set(
        v.x * cos(ang) - v.z * sin(ang),
        v.y,
        v.x * sin(ang) + v.z * cos(ang)
    );
}

vector line_dir = normalize(rotate2D(set(1, 0, 0), line_angle));
vector place_dir = normalize(rotate2D(set(1, 0, 0), place_angle));

float bounds[] = detailintrinsic(1, "bounds");
vector min_bound = set(bounds[0], bounds[2], bounds[4]);
vector max_bound = set(bounds[1], bounds[3], bounds[5]);
float max_length = length(min_bound - max_bound) * 2;

int line_num = int(max_length / base_interval);
vector base_slider = place_dir * max_length / line_num;
vector center_pos = detail(1, "centroid");

for(int i = 0; i < line_num; ++i)
{
    vector line_center_pos = center_pos - max_length / 2 * place_dir;
    line_center_pos += base_slider / 2;
    vector randomly_slide_dir = normalize(base_slider) * interval_range;
    randomly_slide_dir *= rand(base_slider * i + line_center_pos) - 0.5;
    line_center_pos += base_slider * i + randomly_slide_dir;
    
    vector uv;
    vector pt_1_pos;
    vector pt_2_pos;
    int hit_prim_1 = intersect(2, line_center_pos, line_dir * max_length, pt_1_pos, uv);
    int hit_prim_2 = intersect(2, line_center_pos, -line_dir * max_length, pt_2_pos, uv);    
    if(hit_prim_1 < 0 || hit_prim_2 < 0) continue;
    
    int pt_1 = addpoint(0, pt_1_pos);
    int pt_2 = addpoint(0, pt_2_pos);        
    int polyline = addprim(0, "polyline");
    addvertex(0, polyline, pt_1);
    addvertex(0, polyline, pt_2);
}


■BaseLineGenerator
○BottomStartPointも生成する
・角度の逆転対応

●ソース
float rotate_angle = ch("RotateAngle");
float interval = ch("Interval");

float calculating_length = 0;
vector side_start_pos_list[];
vector bottom_start_pos_list[];
int bottom_line_pt_list[] = primpoints(0, @primnum);
int top_line_pt_list[] = primpoints(0, i@TopLinePrimNum);

//側辺で生成
int side_pt_1 = top_line_pt_list[0];
int side_pt_2 = bottom_line_pt_list[0];
if(rotate_angle < 0)
{
    side_pt_1 = bottom_line_pt_list[len(bottom_line_pt_list) - 1];
    side_pt_2 = top_line_pt_list[len(top_line_pt_list) - 1];
}
vector side_pt_1_pos = point(0, "P", side_pt_1);
vector side_pt_2_pos = point(0, "P", side_pt_2);
float side_length = length(side_pt_2_pos - side_pt_1_pos);
vector side_dir = normalize(side_pt_2_pos - side_pt_1_pos);

int loop_num = 0;
while(calculating_length < side_length)
{
    if(side_length < calculating_length + interval / abs(sin(radians(rotate_angle)))) break;
    calculating_length += interval / abs(sin(radians(rotate_angle)));
    vector spawn_pos = side_pt_1_pos + calculating_length * side_dir;    
    append(side_start_pos_list, spawn_pos);
    
    loop_num += 1;
    if(1000 < loop_num) break;
}

calculating_length = calculating_length - side_length;

//底辺で生成
for(int i = 1; i < len(bottom_line_pt_list); ++i)
{
    int last_prim_pt = bottom_line_pt_list[i - 1];
    int prim_pt = bottom_line_pt_list[i];
    vector last_prim_pt_pos = point(0, "P", last_prim_pt);
    vector prim_pt_pos = point(0, "P", prim_pt);
    float bottom_length = length(prim_pt_pos - last_prim_pt_pos);
    vector bottom_dir = normalize(prim_pt_pos - last_prim_pt_pos);

    loop_num = 0;
    while(calculating_length < bottom_length)
    {
        if(bottom_length < calculating_length + interval / abs(cos(radians(rotate_angle)))) break;    
        calculating_length += interval / abs(cos(radians(rotate_angle)));
        vector spawn_pos = last_prim_pt_pos + calculating_length * bottom_dir;
        append(bottom_start_pos_list, spawn_pos);        
        
        loop_num += 1;
        if(1000 < loop_num) break;
    }        
        
    calculating_length = calculating_length - bottom_length;
}

foreach(vector spawn_pos; side_start_pos_list){
    addpoint(0, spawn_pos);
}
foreach(vector spawn_pos; bottom_start_pos_list){
    addpoint(0, spawn_pos);
}
