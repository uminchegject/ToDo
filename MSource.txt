vector base_pos = 0;
vector base_scale = 1;
float height = 1;
int toggle = 1;

for(int i = 0; i < 10; i++)
{
    base_scale *= 0.9;
    height *= 1.2;

    if(toggle == 0){
        vector scale = base_scale;
        scale.x *= rand(scale) * 0.2 + 0.4;
        scale.z *= rand(scale) * 1 + 0.6;
        scale.y = height;

        vector block1_pos = base_pos;
        block1_pos.x += base_scale.x / 2;
        block1_pos.y -= height / 2;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);

        vector block2_pos = base_pos;
        block2_pos.x -= base_scale.x / 2;
        block2_pos.y -= height / 2;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);

        toggle = 1;
    }
    else if(toggle == 1){
        vector scale = base_scale;
        scale.z *= rand(scale) * 0.2 + 0.4;
        scale.x *= rand(scale) * 1 + 0.6;
        scale.y = height;

        vector block1_pos = base_pos;
        block1_pos.z += base_scale.z / 2;
        block1_pos.y -= height / 2;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);

        vector block2_pos = base_pos;
        block2_pos.z -= base_scale.z / 2;
        block2_pos.y -= height / 2;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);

        toggle = 2;
    }    
    else{    
        vector scale = base_scale;
        scale.z *= rand(scale) * 0.2 + 0.4;
        scale.x *= rand(scale) * 0.2 + 0.4;
        scale.y = height;
    
        vector block1_pos = base_pos;
        block1_pos.z += base_scale.z / 2;
        block1_pos.x += base_scale.x / 2;
        block1_pos.y -= height / 2;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);
    
        vector block2_pos = base_pos;
        block2_pos.z += base_scale.z / 2;
        block2_pos.x -= base_scale.x / 2;
        block2_pos.y -= height / 2;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);
                
        vector block3_pos = base_pos;
        block3_pos.z -= base_scale.z / 2;
        block3_pos.x += base_scale.x / 2;
        block3_pos.y -= height / 2;
        int block3 = addpoint(0, block3_pos);
        setpointattrib(0, "scale", block3, scale);              
    
        vector block4_pos = base_pos;
        block4_pos.z -= base_scale.z / 2;
        block4_pos.x -= base_scale.x / 2;
        block4_pos.y -= height / 2;
        int block4 = addpoint(0, block4_pos);
