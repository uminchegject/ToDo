全体の構成を単一のモデリング工程に依存させるのではなく、

特徴的な折板状の空間構成やリズミカルな陰影を再現しています。
これにより、形状のバリエーションやスケールの変化に柔軟に対応しつつ、
構造的な特徴を保ったまま効率的に形態の生成と検証を行える作品となっています。

vector base_pos = v@P;
vector normal = v@N
vector base_scale = v@scale;
float height = base_scale.y;

float block_height = 0
for(int i = 0; i < chi("param_list"); ++i)
{
	float pattern_size = chi("pattern_size")
	block_height += pattern_size;	
}

int block_num = int(height / block_height);
float adjust_block_height = height / block_num;

vector pos = base_pos;
pos.y -= height / 2;
for(int block_idx = 0; block_idx < block_num; ++ block_num)
{
	for(int param_idx = 0; param_idx < chi("param"); ++param_idx)
	{
	
		float pattern_size = ch("pattern_size" + itoa(param_idx + 1));
		float pattern_height = ch("pattern_height" + itoa(param_idx + 1));
		
		pos.y += (pattern_size / block_height) * adjust_block_height / 2;
		
		vector spawn_pos = pos;
		spawn_pos += normal * pattern_height;
		vector scale = base_scale;
		scale.y = pattern_size;
		int pt = addpoint(0, spawn_pos);
		setpointattrib(0, "scale", pt, scale);
		setpointattrib(0, "N", pt, normal);
		
		pos.y += (pattern_size / block_height) * adjust_block_height / 2;
	}
}


●BasePointGenerator
・パラメーター
 -ユニットスケール
 -color
・Attribute
 -scale
 -color
 -Normal

vector unit_scale = chv("unit_scale");
vector color = chv("color");

vector base_pos = v@P;
vector base_scale = v@scale;
vector normal = v@N;
vector up = set(0, 1, 0);
vector tangent = normalize(cross(normal, up));

int width_num = int(base_scale.x / unit_scale.x);
int height_num = int(base_scale.y / unit_scale.y);

vector adjusted_unit_scale = unit_scale;
adjusted_unit_scale.x = base_scale.x / width_num;
adjusted_unit_scale.y = base_scale.y / height_num;

for(int width_idx = 0; width_idx < width_num; width_idx++)
{
	for(int height_idx = 0; height_idx < height_num; height_idx++)
	{
		vector pos = base_pos;
		pos -= tangent * base_scale.x / 2;
		pos -= up * base_scale.y / 2;
		
		pos += unit_scale.x * tangent;
		pos += unit_scale.y * up;
		
		int pt = addpoint(0, pos);
		setpointattrib(0, "scale", pt, adjusted_unit_scale);
		setpointattrib(0, "N", pt, normal);
		setpointattrib(0, "Cd", pt, color);		
	}
}
