vector base_pos;
vector base_scale;
float height;
toggle;

base_scale *= 0.8;
height *= 1.2;

if(toggle == 0){
	vector scale = base_scale;
	scale.x *= rand() * 0.2 + 0.4;
	scale.y = height

	vector block1_pos = base_pos;
	block1_pos.x += base_scale.x / 2;
	int block1 = addpoint(0, block1_pos);
	setpointattrib(0, "scale", block1, scale);

	vector block2_pos = base_pos;
	block2_pos.x -= base_scale.x / 2;
	int block2 = addpoint(0, block2_pos);
	setpointattrib(0, "scale", block2, scale);
}
