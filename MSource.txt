void block_base_generator(vector pos; vector normal; vector scale; int height_unit_num)
{

	float pipe_length = 0;
	int block_num = 0;
	int toggle = 0;	
	for(int height_idex = 0; height_idex < height_unit_num; height_idex++)
	{	
		
		pos.y += scale.y;
		
		block_num += 1;
		if(block_num < rand(pos) * 5 + 3){
			toggle = !toggle;
			block_num = 0;
		}
		
		if(toggle)
		//Pipe
		{
			pipe_length += scale.y;

			vector pipe_block_scale = scale;
			pipe_block_scale.y = pipe_length;
			int pt = addpoint(0, pos);
			setpointattrib(0, "N", pt, normal);
			setpointattrib(0, "scale", pt, pipe_block_scale);
		}
		else
		//Cupsule
		{
			int pt = addpoint(0, pos);
			setpointattrib(0, "N", pt, normal);
			
			
			
			pipe_length = 0;
		}
	}
}
