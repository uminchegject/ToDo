●boundの取得
vector get_bound(vector points[]; vector axis)
{
    axis = normalize(axis);
    float minProj = 1e10;
    float maxProj = -1e10;

    foreach (vector P; points)
    {
        float proj = dot(P, axis);
        minProj = min(minProj, proj);
        maxProj = max(maxProj, proj);
    }

    float size = maxProj - minProj;
    return set(minProj, maxProj, size);
}

int on_line( vector p; int edge)
{
  vector a = point(0, "P",hedge_srcpoint(0, edge));
  vector b = point(0, "P",hedge_dstpoint(0, edge)); 
   
  vector pa = p - a, ba = b - a;
  float h = clamp( dot(pa,ba)/dot(ba,ba), 0.0, 1.0 );
  return  length(pa - ba*h) < 1e-9;
}
