float bounds[] = primintrinsic(0, "bounds", @primnum);
vector min_bound = set(bounds[0], 0, bounds[4]);
vector max_bound = set(bounds[1], 0, bounds[5]);
vector base_scale = set(0, 0, 0);
base_scale.x = length(max_bound - min_bound);
base_scale.y = bounds[3] - bounds[1];
vector x_axis_dir = normalize(max_bound - min_bound);
vector y_axis_dir = set(0, 1, 0);

int block_num_x = int(base_scale.x / block_scale.x);
int block_num_y = int(base_scale.y / block_scale.y);
vector fixed_block_scale.x = {1, 1, 1};
fixed_block_scale.x = base_scale.x / (base_scale.x / block_num_x)
fixed_block_scale.y = base_scale.y / (base_scale.y / block_num_y)

vector spawn_base_pos = v@P;
spawn_base_pos -= x_axis_dir * base_scale.x / 2 + fixed_block_scale.x / 2;
spawn_base_pos -= y_axis_dir * base_scale.y / 2 + fixed_block_scale.y / 2;

for(int x_index = 0; block_num_x; ++x_index)
{
	for(int y_index = 0; block_num_y; ++y_index)
	{
		vector spawn_pos = spawn_base_pos;
		spawn_pos += x_axis_dir * x_index * fixed_block_scale.x;
		spawn_pos += y_axis_dir * y_index * fixed_block_scale.y;
		
		int pt = addpoint(spawn_pos, 0);
	}
}
