void generate_base_point(vector dir; vector pos; float unit_size; int unit_num; float pattern_list[])
{
    vector up = {0, 1, 0};
    vector fwd = normalize(dir);
    vector tangent = normalize(cross(fwd, up));
    
    int pattern_len = len(pattern_list);
    
    for (int unit_idx = 0; unit_idx < unit_num; unit_idx++)
    {
        for (int pattern_idx = 0; pattern_idx < pattern_len; pattern_idx++)
        {
            float pattern = pattern_list[pattern_idx];
            
            vector spawn_pos = pos
                + fwd * unit_size * (unit_idx * pattern_len + pattern_idx)
                + tangent * unit_size * pattern;
            
            int pt = addpoint(0, spawn_pos);
            
            setpointattrib(0, "scale", pt, set(unit_size, unit_size, unit_size * 2), "set");
            setpointattrib(0, "N", pt, tangent, "set");
        }
    }
}

float bounds[] = primintrinsic(0, "bounds", @primnum);
vector bmin = set(bounds[0], bounds[2], bounds[4]);
vector bmax = set(bounds[1], bounds[3], bounds[5]);
vector base_pos = (bmin + bmax) * 0.5;
vector base_scale = bmax - bmin;

float unit_size = ch("unit_size");
int unit_num_x = int(base_scale.x / (unit_size * 4)) * 4;
int unit_num_y = int(base_scale.y / (unit_size * 4)) * 4;
int unit_num_z = int(base_scale.z / (unit_size * 4)) * 4;

vector adjusted_base_scale;
adjusted_base_scale.x = unit_num_x * unit_size;
adjusted_base_scale.y = unit_num_y * unit_size;
adjusted_base_scale.z = unit_num_z * unit_size;


float pattern_list[] = array(0, -1, -1, 0);

for(int unit_idx_y = 0; unit_idx_y < unit_num_y; unit_idx_y++){

    vector pos = base_pos;
    pos.y += unit_size * (unit_idx_y + 0.5) - base_scale.y / 2;
    pos.x -= adjusted_base_scale.x / 2 - unit_size / 2;        
    pos.z -= adjusted_base_scale.z / 2 - unit_size;  
    
    generate_base_point({1, 0, 0}, pos, unit_size, unit_num_x / 4, pattern_list);

    pos.x += adjusted_base_scale.x - unit_size * 1.5;
    pos.z -= unit_size / 2;    
    generate_base_point({0, 0, 1}, pos, unit_size, unit_num_z / 4, pattern_list);

    pos.z += adjusted_base_scale.z - unit_size * 1.5;
    pos.x += unit_size / 2;
    generate_base_point({-1, 0, 0}, pos, unit_size, unit_num_x / 4, pattern_list);
    
    pos.x -= adjusted_base_scale.x - unit_size * 1.5;
    pos.z += unit_size / 2;    
    generate_base_point({0, 0, -1}, pos, unit_size, unit_num_z / 4, pattern_list);
    
}
