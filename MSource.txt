vector base_scale = v@scale;

float unit_size = base_scale.x / 6;

float height_unit_size = ch("height_unit_size");
int height_unit_num = int(base_scale.y / height_unit_size);
height_unit_size = base_scale.y / height_unit_num;

for(int idx = 0; idx < height_unit_num; idx++)
{
    vector pos = v@P;
    pos.y += height_unit_size * idx;
    
    //Frame
    vector frame_scale = set(unit_size * 2, height_unit_size, unit_size * 2);
    int frame_pt = addpoint(0, pos);
    setpointattrib(0, "scale", frame_pt, frame_scale);
    
    //Panel
    if(int(idx / 5) % 2 == 0){
        vector panel_scale = set(unit_size * 2, height_unit_size, unit_size * 2);
        int panel_pt = addpoint(0, pos);
        setpointattrib(0, "scale", panel_pt, panel_scale);
        setpointgroup(0, "panel", panel_pt, 1);
    }
}
removepoint(0, @ptnum);


●配置処理(カプセル・PipePrim)
void block_base_generator(vector pos; vector normal; vector scale; int height_unit_num)
{

	float pipe_length = 0;
	int block_num = 0;
	int toggle = 0;	
	for(int height_idex = 0; height_idex < height_unit_num; height_idex++)
	{	
		
		pos.y += scale.y;
		
		block_num += 1;
		if(block_num < 5){
			toggle = !toggle;
			block_num = 0;
		}
		
		if(toggle)
		//Pipe
		{
			pipe_length += scale.y;

			vector pipe_block_scale = scale;
			pipe_block_scale.y = pipe_length;
			int pt = addpoint(0, pos);
			setpointattrib(0, "N", pt, normal);
			setpointattrib(0, "scale", pt, pipe_block_scale);
		}
		else
		//Cupsule
		{
			int pt = addpoint(0, pos);
			setpointattrib(0, "N", pt, normal);
			
			
			
			pipe_length = 0;
		}
	}
}

// inspired by https://iquilezles.org/www/articles/distfunctions/distfunctions.htm
int on_line(vector p; int edge)
{
  vector a = point(0, "P",hedge_srcpoint(0, edge));
  vector b = point(0, "P",hedge_dstpoint(0, edge)); 
   
  vector pa = p - a, ba = b - a;
  float h = clamp( dot(pa,ba)/dot(ba,ba), 0.0, 1.0 );
  return  length(pa - ba*h) < 1e-9;
}

vector base_pos = 0; //v@P;
vector base_scale = {1.0, 10.0, 1.0};

int center_block_num = chi("center_block_num");
float center_ratio = ch("center_ratio");
float center_height_ratio = ch("center_height_ratio");

//Center
vector center_scale = base_scale;
center_scale.x *= center_ratio;
center_scale.z *= center_ratio;

for(float idx = 0; idx < center_block_num; idx++)
{
    vector block_scale = center_scale;
    block_scale.x *= (1 - idx / center_block_num);
    block_scale.z *= (1 - idx / center_block_num);
    block_scale.y *= (1 - center_height_ratio) * (idx / center_block_num) + center_height_ratio;
    
    int pt = addpoint(0, base_pos);
    setpointattrib(0, "scale", pt, block_scale);
}

//Side
int side_block_num = chi("side_block_num");

int toggle = 1;
for(float idx = 0; idx < side_block_num; idx++)
{
    vector block_scale = base_scale;
    block_scale.y *= center_height_ratio;
    
    block_scale.x *= (1 - center_ratio) * (idx / side_block_num) + center_ratio;
    block_scale.z *= (1 - center_ratio) * (idx / side_block_num) + center_ratio;
    
    block_scale.y *= (1 - (idx / side_block_num) * (rand(idx + 10000) * 0.6 + 0.7));

    if(toggle == 0){
        
        vector scale = block_scale;
        float offset;
        
        if(int(idx) % 6 != 2){            
            scale.x *= 0.3;
            offset = (1 - (scale.x / block_scale.x)) / 2 * block_scale.x;
        }
        else{
            //scale.z *= 1.1;
            scale.x *= 0.2;
            offset = ((scale.x / block_scale.x)) / 2 * block_scale.x;        
        }        

        vector block1_pos = base_pos;
        block1_pos.x += offset;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);

        vector block2_pos = base_pos;
        block2_pos.x -= offset;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);

        toggle = 1;
    }   
    else if(toggle == 1){
        vector scale = block_scale;
        float offset;
        if(int(idx) % 6 != 0){            
            scale.z *= 0.3;
            offset = (1 - (scale.z / block_scale.z)) / 2 * block_scale.z;
        }
        else{
            //scale.x *= 1.1;
            scale.z *= 0.15;
            offset = ((scale.z / block_scale.z)) / 2 * block_scale.z;        
        }         

        vector block1_pos = base_pos;
        block1_pos.z += offset;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);

        vector block2_pos = base_pos;
        block2_pos.z -= offset;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);

        toggle = 2;
    }
    else{    
        vector scale = block_scale;
        scale.z *= 0.2;
        scale.x *= 0.2;
        float offset_x = (1 - (scale.x / block_scale.x)) / 2 * block_scale.x;           
        float offset_z = (1 - (scale.z / block_scale.z)) / 2 * block_scale.z;           
    
        vector block1_pos = base_pos;
        block1_pos.x += offset_x;
        block1_pos.z += offset_z;
        int block1 = addpoint(0, block1_pos);
        setpointattrib(0, "scale", block1, scale);
    
        vector block2_pos = base_pos;
        block2_pos.x -= offset_x;
        block2_pos.z += offset_z;
        int block2 = addpoint(0, block2_pos);
        setpointattrib(0, "scale", block2, scale);
                                        
        vector block3_pos = base_pos;
        block3_pos.x += offset_x;
        block3_pos.z -= offset_z;
        int block3 = addpoint(0, block3_pos);
        setpointattrib(0, "scale", block3, scale);              
    
        vector block4_pos = base_pos;
        block4_pos.x -= offset_x;
        block4_pos.z -= offset_z;
        int block4 = addpoint(0, block4_pos);
        setpointattrib(0, "scale", block4, scale);
    
        toggle = 0;
    }    
}
