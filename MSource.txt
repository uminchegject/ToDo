for(int plane_index = 0; plane_index < plane_num; ++plane_index)
{	
	if(i % 3)
	{
		vector4 plane = make_plane();
		vector base_point_pos = get_intersect_pos(start_pt_pos, end_pt_pos);
		
		if(length(base_point_pos) == 0)continue;
		
		if(length(base_point_pos - start_pt_pos) < block_length){
			scale = length(base_point_pos - start_pt_pos);
		}		
		else if(length(base_point_pos - end_pt_pos) < block_length){
			scale = length(base_point_pos - end_pt_pos);
		}
		else{
			scale = block_length;
		}
	}
}
