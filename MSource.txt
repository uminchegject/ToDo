void generate_base_point(vector dir, vector pos, float unit_scale, int unit_num, float pattern_list[])
{
	vector up = {0, 1, 0};
	vector tangent = normalize(dir, up);
	
	for (int unit_idx = 0; unit_idx < unit_num; unit_idx++)
	{
		vector spawn_pos = pos + dir * len(pattern_list) * unit_scale * unit_idx;
		for(int pattern_idx = 0; pattern_idx < len(pattern_list); pattern_idx++)
		{
			spawn_pos += dir * unit_scale * pattern_idx;
		
			float pattern = pattern_list[pattern_idx];
			spawn_pos += tangent * unit_scale * pattern;
			int pt = addpoint(0, spawn_pos);
			setpointattrib(0, "N", pt, tangent);
			
		}
	}
}
