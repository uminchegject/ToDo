function vector get_points_intersect_pos(vector p0, p1, p2, p3)
{
    vector u = p1 - p0;
    vector v = p3 - p2;
    vector w = p0 - p2;

    float a = dot(u,u);
    float b = dot(u,v);
    float c = dot(v,v);
    float d = dot(u,w);
    float e = dot(v,w);

    float sc = (b*e - c*d) / (a*c - b*b);
    float tc = (a*e - b*d) / (a*c - b*b);    

    vector pA = p0 + sc * u;
    vector pB = p2 + tc * v;

    return (pA + pB) * 0.5;
}

vector4 make_plane(vector pos; vector normal)
{
    vector n = normalize(normal);
    float d = -dot(n, pos);
    return set(n.x, n.y, n.z, d);
}

vector get_plane_intersect_pos(vector p1; vector p2; vector4 plane)
{
    vector pos1 = p1;
    vector pos2 = p2;
    int result;
    clip(result, pos1, pos2, plane);
        
    vector intersect = {0,0,0};
    if (result == 1)
    {
        if (pos1 != p1)
            intersect = pos1;
        else if (pos2 != p2)
            intersect = pos2;
        else
            return {0,0,0};
    }
    else
    {
        return {0,0,0};
    }
        
    return intersect;
}

vector get_normal(vector p0; vector p1; vector p2)
{
    vector e1 = p1 - p0;
    vector e2 = p2 - p0;
    return normalize(cross(e1, e2));
}

    base_plan_pt1_pos,
    base_plan_pt1_pos + base_plan_depth_vec * 10000,

    plan_base_pos + center_edge_dir * 10000);
