function vector lineIntersect3D(vector p0, p1, p2, p3)
{
    vector u = p1 - p0;
    vector v = p3 - p2;
    vector w = p0 - p2;

    float a = dot(u,u);
    float b = dot(u,v);
    float c = dot(v,v);
    float d = dot(u,w);
    float e = dot(v,w);

    float sc = (b*e - c*d) / (a*c - b*b);
    float tc = (a*e - b*d) / (a*c - b*b);    

    vector pA = p0 + sc * u;
    vector pB = p2 + tc * v;

    return (pA + pB) * 0.5;
}


//パラメーター
float base_rotate = ch("BaseRotate");
float base_height = ch("BaseHeight");
float base_rotate_height = ch("BaseRotateHeight");

vector axis = set(0, 1, 0);
matrix3 rot = ident();
vector last_rotate_vector;
int last_pt = 0;

int prim_pt_list[] = primpoints(0, i@primnum);
int new_pt_list[];


for(int i = 0; i < len(prim_pt_list); ++i)
{
    int last_prim_pt = prim_pt_list[i - 1];
    int prim_pt = prim_pt_list[i];
    int next_prim_pt = prim_pt_list[i + 1];
    
    vector last_prim_pt_pos = point(0, "P", last_prim_pt);
    vector prim_pt_pos = point(0, "P", prim_pt);
    vector next_prim_pt_pos = point(0, "P", next_prim_pt);
    
    vector last_prim_pt_dir = normalize(last_prim_pt_pos - prim_pt_pos);
    vector next_prim_pt_dir = normalize(next_prim_pt_pos - prim_pt_pos);
    
        
        //パラメーター
    float ramp_pos = float(i) / len(prim_pt_list);
    float rotate_ramp = chramp("rotate_param", ramp_pos);
    float height_ramp = chramp("height_param", ramp_pos);
    float rotate_height_ramp = chramp("rotate_height_param", ramp_pos);
        
    float rotate_angle = rotate_ramp * base_rotate;
    float height = height_ramp * base_height;
    float rotate_height = rotate_height_ramp * base_rotate_height;
    
    //回転処理
    if(0 < i)
    {               
        rot = ident();
        rotate(rot, radians(rotate_angle), axis);
        vector rotate_vector = rot * (prim_pt_pos - last_prim_pt_pos);
        prim_pt_pos = last_prim_pt_pos + rotate_vector;
        setpointattrib(0, "P", prim_pt, prim_pt_pos);
    }     
        
    vector pt_pos = prim_pt_pos;

    //傾き
    if(0 == i)
    {
        rot = ident();
        rotate(rot, radians(90), axis);
        vector rotate_dir = next_prim_pt_dir * rot;
        vector rotate_vector = rotate_dir * rotate_height * tan(radians(height));
        pt_pos += rotate_vector;
        last_rotate_vector = rotate_vector;
    }
    else if(i == len(prim_pt_list) - 1)
    {
        pt_pos += last_rotate_vector;
    }
    else
    {
        rot = ident();
        rotate(rot, radians(90), axis);
        vector rotate_dir = next_prim_pt_dir * rot;
        vector rotate_vector = rotate_dir * rotate_height * tan(radians(height));
        
        vector intersect_pos = lineIntersect3D(
            last_prim_pt_pos + last_rotate_vector,
            prim_pt_pos + last_rotate_vector,
            prim_pt_pos + rotate_vector,
            next_prim_pt_pos + rotate_vector);
        pt_pos = intersect_pos;
        
        last_rotate_vector = rotate_vector;
    }
        
    //高さ調整
    //pt_pos.y += height;
    
    int pt = addpoint(0, pt_pos);
    if(0 < i)
    {
        int prim = addprim(0, "poly");
        addvertex(0, prim, prim_pt);
        addvertex(0, prim, pt);
        addvertex(0, prim, last_pt);
        addvertex(0, prim, last_prim_pt);
    }
    last_pt = pt;
}


removeprim(0, @primnum, 1);
