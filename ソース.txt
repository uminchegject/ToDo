■PompidouPattern
vector base_scale = chv("base_scale");
vector base_unit_scale = chv("base_unit_scale");
int x_skip_num = chi("x_skip_num");
int y_skip_num = chi("y_skip_num");

vector base_pos = 0;

int unit_num_x = int(base_scale.x / base_unit_scale.x);
int unit_num_y = int(base_scale.y / base_unit_scale.y);

vector unit_scale = 1;
unit_scale.x = base_scale.x / float(unit_num_x);
unit_scale.y = base_scale.y / float(unit_num_y);

for(int x_idx = 0; x_idx < unit_num_x; x_idx++){

    for(int y_idx = 0; y_idx < unit_num_y; y_idx++){
        
        if(x_idx % x_skip_num != 0) continue;
        if(y_idx % y_skip_num != 0) continue;
                
        vector pos = base_pos;
        pos.x += x_idx * unit_scale.x;
        pos.y += y_idx * unit_scale.y;
        int pt = addpoint(0, pos);
        setpointattrib(0, "scale", pt, unit_scale);     
    }
}
