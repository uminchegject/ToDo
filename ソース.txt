vector base_scale;
vector base_pos;
vector base_unit_scale;

int unit_num_x = (base_scale.x / base_unit_scale.x);
int unit_num_y = (base_scale.y / base_unit_scale.y);

vector unit_scale = 1;
unit_scale.x = base_scale / unit_num_x;
unit_scale.y = base_scale / unit_num_y;

for(int x_idx = 0; x_idx < unit_num_x; x_idx++){

    for(int y_idx = 0; y_idx < unit_num_y; y_idx++){
        vector pos = base_pos;
        pos.x += x_idx * unit_scale.x;
        pos.y += y_idx * unit_scale.y;
        
        //Frame
        vector frame_pos = pos;
        vector frame_scale = unit_scale * ch("frame_scale");
        int frame_pt = addpoint(0, frame_pos);
        setpointattrib(0, "scale", frame_pt, frame_scale);

        //Box
        vector box_pos = pos;
        vector box_scale = unit_scale * ch("box_scale");
        int box_pt = addpoint(0, box_pos);
        setpointattrib(0, "scale", box_pt, box_scale);

        //Pipe
        int pipe_num = 0;
        for(int pipe_idx = 0; pipe_idx < pipe_num; ++pipe_idx){
            vector pipe_scale = unit_scale;
            vector pipe_pos = pos;

            int pipe_pt = addpoint(0, pipe_pos);
            setpointattrib(0, "scale", pipe_pt, pipe_unit_scale);
        }       
    }
}
